# Saty Phase Oscillator
# Copyright (C) 2022-2025 Saty Mahajan
# A useful range-based signal to monitor various phases of the market.

declare lower;
input Time_Warp = {default "off", "1m", "2m", "3m", "4m", "5m", "10m", "15m", "20m", "30m", "1h", "2h", "4h", "D", "W", "M", "Y"};
input Show_Labels = yes;
input Show_Zone_Crosses = yes;
input Show_Extreme_Crosses = yes;
input Compression_Colors = {default Clean, Classic};

# Saty Color Theme
DefineGlobalColor("Saty Green", CreateColor(0,255, 30));
DefineGlobalColor("Saty Blue", CreateColor(0, 185, 255));
DefineGlobalColor("Saty Red", CreateColor(255, 0, 0));
DefineGlobalColor("Saty Yellow", CreateColor(255,255,0));
DefineGlobalColor("Saty Orange", CreateColor(255, 150, 0));
DefineGlobalColor("Saty Violet", CreateColor(150, 100, 255));
DefineGlobalColor("Saty Purple", CreateColor(150, 0, 150));
DefineGlobalColor("Saty Pink", CreateColor(255, 0, 255));
DefineGlobalColor("Saty White", CreateColor(255, 255, 255));
DefineGlobalColor("Saty Light Gray", CreateColor(200,200,200));
DefineGlobalColor("Saty Gray", CreateColor(150,150,150));
DefineGlobalColor("Saty Dark Gray", CreateColor(100,100,100));
DefineGlobalColor("Saty Black", CreateColor(0, 0, 0));

# Compression Color Mode
def compression_color_mode = if Compression_Colors is equal to Compression_Colors.Clean then 1 else if Compression_Colors is equal to Compression_Colors.Classic then 0 else 0;

# Time Warp
def period;
switch (Time_Warp) {
case "off":
    period = GetAggregationPeriod();
case "1m":
    period = AggregationPeriod.MIN;
case "2m":
    period = AggregationPeriod.TWO_MIN;
case "3m":
    period = AggregationPeriod.THREE_MIN;
case "4m":
    period = AggregationPeriod.FOUR_MIN;
case "5m":
    period = AggregationPeriod.FIVE_MIN;
case "10m":
    period = AggregationPeriod.TEN_MIN;
case "15m":
    period = AggregationPeriod.FIFTEEN_MIN;
case "20m":
    period = AggregationPeriod.TWENTY_MIN;
case "30m":
    period = AggregationPeriod.THIRTY_MIN;
case "1h":
    period = AggregationPeriod.HOUR;
case "2h":
    period = AggregationPeriod.TWO_HOURS;
case "4h":
    period = AggregationPeriod.FOUR_HOURS;
case "D":
    period = AggregationPeriod.DAY;
case "W":
    period = AggregationPeriod.WEEK;
case "M":
    period = AggregationPeriod.MONTH;
case "Y":
    period = AggregationPeriod.YEAR;
}

# Saty Phase Oscillator Signal
def price = close (period = period);
def atr = Round(WildersAverage(TrueRange(high(period = period), close(period = period), low(period = period)), 14), 2);
def pivot = ExpAverage(price, 21);
def above_pivot = price >= pivot;
def raw_signal = ((price - pivot) / (3.0 * atr)) * 100;
def oscillator_signal = Round(ExpAverage(Round(raw_signal, 2),3),2);
plot phase_oscillator = oscillator_signal;
phase_oscillator.SetLineWeight(2);

# Phases
plot extended_up_zone = 100.0;
extended_up_zone.AssignValueColor(Color.LIGHT_GRAY);
extended_up_zone.HideBubble();
extended_up_zone.HideTitle();
plot distribution_zone = 61.8;
distribution_zone.AssignValueColor(Color.GRAY);
distribution_zone.HideBubble();
distribution_zone.HideTitle();
plot neutral_up_zone = 23.6;
neutral_up_zone.AssignValueColor(Color.GRAY);
neutral_up_zone.HideBubble();
neutral_up_zone.HideTitle();
plot neutral_down_zone = -23.6;
neutral_down_zone.AssignValueColor(Color.GRAY);
neutral_down_zone.HideBubble();
neutral_down_zone.HideTitle();
plot accumulation_zone = -61.8;
accumulation_zone.AssignValueColor(Color.GRAY);
accumulation_zone.HideBubble();
accumulation_zone.HideTitle();
plot extended_down_zone = -100.0;
extended_down_zone.AssignValueColor(Color.LIGHT_GRAY);
extended_down_zone.HideBubble();
extended_down_zone.HideTitle();

# Bollinger Band Compression Signal
def bband_offset = 2.0 * STDev(price, 21);
def bband_up = pivot + bband_offset
;
def bband_down = pivot - bband_offset;
def compression_threshold_up = pivot + (2.0 * atr);
def compression_threshold_down = pivot - (2.0 * atr);
def expansion_threshold_up = pivot + (1.854 * atr);
def expansion_threshold_down = pivot - (1.854 * atr);
def compression = if above_pivot then bband_up - compression_threshold_up else compression_threshold_down - bband_down;
def expansion_zone = if above_pivot then bband_up - expansion_threshold_up else expansion_threshold_down - bband_down;
def expansion = compression[1] <= compression[0];
def compression_tracker = if expansion and expansion_zone > 0 then 0 else if compression <= 0 then 1 else 0;
plot compression_level = If(IsNaN(close), Double.NaN, 0);
compression_level.AssignValueColor(if compression_tracker == 1 and compression_color_mode is equal to 1 then GlobalColor("Saty Light Gray") else if compression_tracker == 1 and compression_color_mode is equal to 0 then GlobalColor("Saty Pink") else Color.DARK_GRAY);
compression_level.HideBubble();
compression_level.HideTitle();
compression_level.SetLineWeight(2);

# Compressing / Expanding / Expanded
AddLabel(show_labels == yes and compression_tracker == 1, "Compression ",  if compression_tracker == 1 and compression_color_mode is equal to 1 then GlobalColor("Saty Light Gray")
    else if compression_tracker == 1 and compression_color_mode is equal to 0 then GlobalColor("Saty Pink") else GlobalColor("Saty Light Gray"));

# Mean Reversion PO Crossover Signals
def leaving_accumulation = (oscillator_signal[1] <= -61.8) and (oscillator_signal > -61.8);
def leaving_extreme_down = (oscillator_signal[1] <= -100) and (oscillator_signal > -100);
def leaving_distribution = (oscillator_signal[1] >= 61.8) and (oscillator_signal < 61.8);
def leaving_extreme_up = (oscillator_signal[1] >= 100) and (oscillator_signal < 100);
plot bullish_cross = if Show_Extreme_Crosses and leaving_extreme_down then oscillator_signal[1] - 5
                     else if Show_Zone_Crosses and leaving_accumulation then oscillator_signal[1] - 5  
                     else Double.NaN;
bullish_cross.SetPaintingStrategy(PaintingStrategy.POINTS);
bullish_cross.AssignValueColor(Color.YELLOW);
bullish_cross.SetLineWeight(1);
bullish_cross.HideBubble();
plot bearish_cross = if Show_Extreme_Crosses and leaving_extreme_up then oscillator_signal[1] + 5
                     else if Show_Zone_Crosses and leaving_distribution then oscillator_signal[1] + 5 
                     else Double.NaN;
bearish_cross.SetPaintingStrategy(PaintingStrategy.POINTS);
bearish_cross.AssignValueColor(Color.YELLOW);
bearish_cross.SetLineWeight(1);
bearish_cross.HideBubble();

# Phase Oscillator Plot Colors
phase_oscillator.AssignValueColor(
    if compression_tracker == 1 and compression_color_mode is equal to 1 then GlobalColor("Saty Light Gray")
    else if compression_tracker == 1 and compression_color_mode is equal to 0 then GlobalColor("Saty Pink")
    else if phase_oscillator >= 0 then GlobalColor("Saty Green")
    else GlobalColor("Saty Red")
);

# Scanning Plots
plot is_compressed = compression_tracker;
is_compressed.SetHiding(1);
is_compressed.HideBubble();
is_compressed.HideTitle();

def extended_up = oscillator_signal >= 100;
def in_distribution = oscillator_signal >= 61.8 and oscillator_signal < 100;
def in_mark_up = oscillator_signal > 23.6 and oscillator_signal < 61.8;
def in_launch_box = oscillator_signal >= -23.6 and oscillator_signal <= 23.6;
def in_mark_down = oscillator_signal > -61.8 and oscillator_signal < -23.6;
def in_accumulation = oscillator_signal <= -61.8 and oscillator_signal > -100;
def extended_down = oscillator_signal <= -100;
plot phase = if in_launch_box then 0 else if in_mark_up then 1 else if in_mark_down then -1 else if in_accumulation then -2 else if in_distribution then 2 else if extended_down then -3 else 3;
phase.SetHiding(1);
phase.HideBubble();
phase.HideTitle();
